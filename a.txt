local function listSoundFiles()
    local files = {}
    if isfolder(SOUND_DIR) then
        for _, name in ipairs(listfiles(SOUND_DIR)) do
            local filename = name:match("([^/\\]+)$") or name
            table.insert(files, filename)
        end
    end
    if #files == 0 then
        table.insert(files, SOUND_NAME)
    end
    return files
end

local function passesTeamCheck(player)
    if not Settings.TeamCheck then return true end
    if not player or not player.Team then return false end
    if not LocalPlayer.Team then return false end
    return player.Team ~= LocalPlayer.Team
end

local function passesWallCheck(originPos, targetPart)
    if not Settings.WallCheck then return true end
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = { LocalPlayer.Character }
    params.IgnoreWater = true

    local dir = targetPart.Position - originPos
    local result = workspace:Raycast(originPos, dir, params)
    if not result then return true end
    if targetPart:IsDescendantOf(result.Instance) or result.Instance:IsDescendantOf(targetPart.Parent) then
        return true
    end
    return false
end

local function getContextSettings(context)
    if context == "Knife" then
        return Settings.KnifeAimPart, Settings.KnifeMaxDistance
    end
    return Settings.GunAimPart, Settings.GunMaxDistance
end

local function getClosestTarget(originPos, context)
    local myChar = LocalPlayer.Character
    local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myHRP then return nil, nil, math.huge end

    local aimPart, maxDist = getContextSettings(context)
    local origin = originPos or (myHRP.Position + Vector3.new(0, 1.5, 0))
    local closestPart, closestPlayer, shortestDist = nil, nil, maxDist

    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer and p.Character then
            local char = p.Character
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health > 0 and passesTeamCheck(p) then
                local part = char:FindFirstChild(aimPart)
                    or char:FindFirstChild("Head")
                    or char:FindFirstChild("HumanoidRootPart")
                if part then
                    local dist = (part.Position - origin).Magnitude
                    if dist < shortestDist and passesWallCheck(origin, part) then
                        shortestDist = dist
                        closestPart = part
                        closestPlayer = p
                    end
                end
            end
        end
    end
    return closestPart, closestPlayer, shortestDist
end

TargetInput.Get = function(...)
    local autoAimEnabled = (CurrentAimContext == "Knife")
        and Settings.KnifeAutoAim
        or Settings.GunAutoAim

    if autoAimEnabled then
        local myChar = LocalPlayer.Character
        local origin
        if myChar and myChar:FindFirstChild("HumanoidRootPart") then
            origin = myChar.HumanoidRootPart.Position + Vector3.new(0, 1.5, 0)
        elseif Camera then
            origin = Camera.CFrame.Position
        else
            origin = Vector3.new(0, 0, 0)
        end

        local part = getClosestTarget(origin, CurrentAimContext)
        if part then
            return origin, (part.Position - origin).Unit
        end
    end
    return TargetInput._originalGet(...)
end

local function canShoot(gun)
    if not gun then return false end
    if not gun.Usage then return false end
    if gun.Usage.IsReloading then return false end
    if gun.Usage.Clip <= 0 then return false end
    local fsm = getFSM(gun)
    if not fsm then return false end
    if fsm.State == "Reload" then return false end
    if fsm.State == "Fire" then return false end
    if fsm.State == "Equip" and fsm.Progress and fsm.Progress < 0.2 then
        return false
    end
    return true
end

local function stopBhopTrack()
    if bhop.CurrentTrack then
        pcall(function()
            bhop.CurrentTrack:Stop(0)
            bhop.CurrentTrack:Destroy()
        end)
        bhop.CurrentTrack = nil
    end

local function playBhop(character)
    if not bhop.Enabled or not character then return end

    bhop.CurrentCharacter = character
    stopBhopTrack()

    local humanoid = character:WaitForChild("Humanoid", 10)
    if not humanoid or character ~= LocalPlayer.Character or not bhop.Enabled then
        return
    end

    local animator = humanoid:FindFirstChildOfClass("Animator")
    if not animator then
        animator = Instance.new("Animator")
        animator.Parent = humanoid
    end

    local animation = Instance.new("Animation")
    animation.AnimationId = BHOP_ANIMATION_ID

    local ok, trackOrError = pcall(function()
        local track = animator:LoadAnimation(animation)
        track.Priority = Enum.AnimationPriority.Action4
        track.Looped = true
        track:Play()
        return track
    end)
    animation:Destroy()

    if ok then
        bhop.CurrentTrack = trackOrError
    end
end

local function playSound()
    if not sound.Enabled or sound.TrackName == "" then
        stopSound()
        return
    end

    local path = SOUND_DIR .. "/" .. sound.TrackName
    if not isfile(path) then
        stopSound()
        return
    end

    stopSound()

    local ok, assetId = pcall(function()
        return getcustomasset(path)
    end)

    if not ok or not assetId then
        return
    end

    local soundObj = Instance.new("Sound")
    soundObj.Name = "MVS2_Sound"
    soundObj.SoundId = assetId
    soundObj.Volume = 1
    soundObj.Looped = true
    soundObj.Parent = workspace.CurrentCamera or workspace

    pcall(function()
        soundObj:Play()
    end)

    sound.Sound = soundObj
end

    if not backward.Enabled then return end

    local Character = LocalPlayer.Character
    local Humanoid = Character and Character:FindFirstChildOfClass("Humanoid")
    local HRP = Character and Character:FindFirstChild("HumanoidRootPart")

    if Humanoid and HRP then
        Humanoid.AutoRotate = false

        local isShiftLock = UserInputService.MouseBehavior == Enum.MouseBehavior.LockCenter

        if isShiftLock then
            local camLook = Camera.CFrame.LookVector
            local flat = Vector3.new(camLook.X, 0, camLook.Z)
            if flat.Magnitude > 0.01 then
                HRP.CFrame = CFrame.lookAt(HRP.Position, HRP.Position - flat.Unit)
            end
        else
            local MoveDirection = Humanoid.MoveDirection
            if MoveDirection.Magnitude > 0 then
                HRP.CFrame = CFrame.lookAt(HRP.Position, HRP.Position - MoveDirection)
            end
        end
    end
end)

    if not walk.Enabled then return end

    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if not hum or not hrp then return end

    if walk.Mode == "Velocity" then
        if hum.MoveDirection.Magnitude > 0 then
            hrp.Velocity = Vector3.new(
                hum.MoveDirection.X * walk.Speed,
                hrp.Velocity.Y,
                hum.MoveDirection.Z * walk.Speed
            )
        end
    else
        if hum.MoveDirection.Magnitude > 0 then
            local move = hum.MoveDirection * walk.Speed * dt
            hrp.CFrame = hrp.CFrame + move
        end
    end
end)

local function sendChatMessage(text)
    if not text or text == "" then return end
    pcall(function()
        local channel = TextChatService.TextChannels:FindFirstChild("RBXGeneral")
        if not channel then
            local ch = TextChatService.TextChannels:GetChildren()
            channel = ch[1]
        end
        if channel then
            channel:SendAsync(text)
        end
    end)
end

RunService.Heartbeat:Connect(function()
    if not autoChat.Enabled then return end
    if #autoChat.Messages == 0 then return end
    local now = tick()
    if now - autoChat.LastSend >= autoChat.Interval then
        autoChat.LastSend = now
        local msg
        if autoChat.Mode == "Random" then
            msg = autoChat.Messages[math.random(#autoChat.Messages)]
        else
            msg = autoChat.Messages[autoChat.Index]
            autoChat.Index = (autoChat.Index % #autoChat.Messages) + 1
        end
        sendChatMessage(msg)
    end
end)

function updateChinaHat()
    if not chinaHatInstance then return end
    chinaHatInstance.Color = chinaHatColor
    chinaHatInstance.Material = Enum.Material[chinaHatMaterial] or Enum.Material.Neon
    local mesh = chinaHatInstance:FindFirstChildOfClass("SpecialMesh")
    if mesh then mesh.Scale = chinaHatScale end

    if weld then weld.C0 = CFrame.new(0, chinaHatOffsetY, 0) end
    local light = chinaHatInstance:FindFirstChildOfClass("PointLight")
    if light then
        light.Color = chinaHatLightColor
        light.Brightness = chinaHatLightBrightness
        light.Range = chinaHatLightRange
    end
end

local function pickDir(lastDir)
    if not lastDir then return HORIZ[math.random(#HORIZ)] end
    local isVertical = math.abs(lastDir.Y) > 0.5
    if not isVertical then
        local r = math.random()
        if r < CFG.UpChance then return Vector3.new(0, 1, 0) end
        if r < CFG.UpChance + CFG.DownChance then return Vector3.new(0, -1, 0) end
    end
    local candidates = {}
    for _, d in ipairs(HORIZ) do
        if math.abs(lastDir:Dot(d)) < 0.1 then table.insert(candidates, d) end
    end
    if #candidates == 0 then return HORIZ[math.random(#HORIZ)] end
    return candidates[math.random(#candidates)]
end

local function makePart(pos)
    local p = Instance.new("Part")
    p.Anchored     = true
    p.CanCollide   = false
    p.Transparency = 1
    p.CastShadow   = false
    p.Size         = Vector3.new(0.1, 0.1, 0.1)
    p.Position     = pos
    p.Parent       = wouldLineFolder
    Instance.new("Attachment", p)
    return p
end

local function makeBeam(pA, pB)
    local b = Instance.new("Beam")
    b.Attachment0    = pA:FindFirstChildOfClass("Attachment")
    b.Attachment1    = pB:FindFirstChildOfClass("Attachment")
    b.Color          = ColorSequence.new(CFG.Color)
    b.Width0         = CFG.Width
    b.Width1         = CFG.Width
    b.FaceCamera     = true
    b.LightEmission  = CFG.LightEmit
    b.LightInfluence = 0
    b.Segments       = 1
    b.Transparency   = NumberSequence.new(0)
    b.Parent         = wouldLineFolder
    return b
end

local function spawnTrail(centerPos)
    local startPos = centerPos + Vector3.new(
        (math.random() - 0.5) * CFG.Range * 2,
        math.random(-5, 10),
        (math.random() - 0.5) * CFG.Range * 2
    )
    local initDir  = pickDir(nil)
    local headPart = makePart(startPos)
    local tailPart = makePart(startPos)
    local beam     = makeBeam(tailPart, headPart)
    wouldTrails[#wouldTrails + 1] = {
        segments = {{ head = headPart, tail = tailPart, beam = beam }},
        headPos  = startPos,
        dir      = initDir,
        traveled = 0,
        born     = tick(),
    }
end

local function removeWouldTrail(t)
    for _, seg in ipairs(t.segments) do
        pcall(function() seg.head:Destroy(); seg.tail:Destroy() end)
    end
end

local function clearWouldLines()
    for i = #wouldTrails, 1, -1 do
        removeWouldTrail(wouldTrails[i])
        table.remove(wouldTrails, i)
    end
end

    Default = "hello",
    Numeric = false,
    Finished = false,
    Text = "Messages (カンマ区切り)",
    Placeholder = "hello, hi, yo",
    Callback = function(v)
        if not v or v == "" then return end
        local msgs = {}
        for part in v:gmatch("[^,]+") do
            local trimmed = part:match("^%s*(.-)%s*$")
            if trimmed ~= "" then table.insert(msgs, trimmed) end
        end
        if #msgs > 0 then
            autoChat.Messages = msgs
            autoChat.Index = 1
        end
    end,
})

-- ===== Update関数 =====
local OptionsTbl = Library.Options
local Toggles = Library.Toggles

local function UpdateColor()
    ColorCorrection.Enabled = Toggles.ColorEnabled.Value
    ColorCorrection.TintColor = OptionsTbl.TintColor.Value
    ColorCorrection.Brightness = OptionsTbl.Brightness.Value
    ColorCorrection.Contrast = OptionsTbl.Contrast.Value
    ColorCorrection.Saturation = OptionsTbl.Saturation.Value
end

local function UpdateFog()
    if Toggles.FogEnabled.Value then
        Lighting.FogStart = OptionsTbl.FogStart.Value
        Lighting.FogEnd = OptionsTbl.FogEnd.Value
        Lighting.FogColor = OptionsTbl.FogColor.Value
    else
        Lighting.FogStart = OriginalFogStart
        Lighting.FogEnd = OriginalFogEnd
        Lighting.FogColor = OriginalFogColor
    end
end

local function UpdateAtmosphere()
    Atmosphere.Parent = Toggles.AtmosphereEnabled.Value and Lighting or nil
    Atmosphere.Density = OptionsTbl.Density.Value
    Atmosphere.Haze = OptionsTbl.Haze.Value
    Atmosphere.Glare = OptionsTbl.Glare.Value
    Atmosphere.Offset = OptionsTbl.Offset.Value
    Atmosphere.Color = OptionsTbl.AtmosphereColor.Value
    Atmosphere.Decay = OptionsTbl.DecayColor.Value
end

Toggles.ColorEnabled:OnChanged(UpdateColor)
OptionsTbl.TintColor:OnChanged(UpdateColor)
OptionsTbl.Brightness:OnChanged(UpdateColor)
OptionsTbl.Contrast:OnChanged(UpdateColor)
OptionsTbl.Saturation:OnChanged(UpdateColor)

Toggles.FogEnabled:OnChanged(UpdateFog)
OptionsTbl.FogStart:OnChanged(UpdateFog)
OptionsTbl.FogEnd:OnChanged(UpdateFog)
OptionsTbl.FogColor:OnChanged(UpdateFog)

Toggles.AtmosphereEnabled:OnChanged(UpdateAtmosphere)
OptionsTbl.Density:OnChanged(UpdateAtmosphere)
OptionsTbl.Haze:OnChanged(UpdateAtmosphere)
OptionsTbl.Glare:OnChanged(UpdateAtmosphere)
OptionsTbl.Offset:OnChanged(UpdateAtmosphere)
OptionsTbl.AtmosphereColor:OnChanged(UpdateAtmosphere)
OptionsTbl.DecayColor:OnChanged(UpdateAtmosphere)

UpdateColor()

local function spawnLandRing(position, delay)
    task.delay(delay, function()
        local ring = Instance.new("Part")
        ring.Name = "LandFx"
        ring.Anchored = true
        ring.CanCollide = false
        ring.CanQuery = false
        ring.CanTouch = false
        ring.Massless = true
        ring.Material = Enum.Material.Neon
        ring.Color = landFxColor
        ring.Transparency = landFxTransparency
        ring.Size = Vector3.new(1, 1, 1)
        ring.CFrame = CFrame.new(position) * CFrame.Angles(
            math.rad(landFxRotationX), math.rad(landFxRotationY), math.rad(landFxRotationZ)
        )
        ring.Parent = Workspace
        local mesh = Instance.new("SpecialMesh")
        mesh.MeshType = Enum.MeshType.FileMesh
        mesh.MeshId = RING_MESH_ID
        mesh.Scale = Vector3.new(landFxSize / 4, landFxSize, landFxSize)
        mesh.Parent = ring
        local startScale = landFxSize
        local endScale = landFxSize * 2
        local t = 0
        local conn
        conn = RunService.Heartbeat:Connect(function(dt)
            t += dt
            local alpha = t / landFxDuration
            if alpha >= 1 then
                ring:Destroy(); conn:Disconnect(); return
            end
            local size = startScale + (endScale - startScale) * alpha
            mesh.Scale = Vector3.new(size / 4, size, size)
            ring.Transparency = landFxTransparency + (1 - landFxTransparency) * alpha
        end)
    end)
end

local function updateLandFx()
    if not landFxEnabled then return end
    local myChar = getMyChar()
    if not myChar then return end
    local hrp = myChar:FindFirstChild("HumanoidRootPart")
    if not hrp then return end

    local velocityY = hrp.AssemblyLinearVelocity.Y

    if velocityY > 2 and not wasFalling then
        wasFalling = true
        local pos = hrp.Position - Vector3.new(0, hrp.Size.Y / 2 + 1 + landFxOffsetY, 0)
        spawnLandRing(pos, 0)
    end

    if wasFalling and velocityY < 0 and math.abs(velocityY) < 1 then
        wasFalling = false
    end
end

function removeTrail()
    if trailInstance then
        local attachParent = trailInstance.Parent
        trailInstance:Destroy()
        trailInstance = nil
        if attachParent then
            for _, name in ipairs({ "TrailAttA", "TrailAttB" }) do
                local a = attachParent:FindFirstChild(name)
                if a then a:Destroy() end
            end
        end
    end
end

local function createTrail()
    removeTrail()
    local myChar = getMyChar()
    if not myChar then return end
    local torsoPart = myChar:FindFirstChild("HumanoidRootPart")
        or myChar:FindFirstChild("UpperTorso")
    if not torsoPart then return end
    local halfLength = trailLength / 2
    local attA = Instance.new("Attachment")
    attA.Name = "TrailAttA"
    attA.Parent = torsoPart
    attA.Position = Vector3.new(0, halfLength, trailOffsetZ)
    local attB = Instance.new("Attachment")
    attB.Name = "TrailAttB"
    attB.Parent = torsoPart
    attB.Position = Vector3.new(0, -halfLength, trailOffsetZ)
    local trail = Instance.new("Trail")
    trail.Attachment0 = attA
    trail.Attachment1 = attB
    trail.Color = ColorSequence.new({
        ColorSequenceKeypoint.new(0, trailColorTop),
        ColorSequenceKeypoint.new(0.5, trailColorMid),
        ColorSequenceKeypoint.new(1, trailColorBottom),
    })
    trail.Lifetime = trailLifetime
    trail.Transparency = NumberSequence.new({
        NumberSequenceKeypoint.new(0, TRAIL_TRANSPARENCY),
        NumberSequenceKeypoint.new(1, 1),
    })
    trail.LightEmission = 1
    trail.LightInfluence = 0
    trail.WidthScale = NumberSequence.new({
        NumberSequenceKeypoint.new(0, TRAIL_WIDTH),
        NumberSequenceKeypoint.new(1, TRAIL_WIDTH),
    })
    trail.Parent = torsoPart
    trailInstance = trail
end

RunService.RenderStepped:Connect(function(dt)
    if not snakeAuraEnabled then
        for _, p in ipairs(snakeAuraParts) do
            if p and p.Parent then p:Destroy() end
        end
        snakeAuraParts = {}
        return
    end
    local myChar = getMyChar()
    local myHRP = myChar and myChar:FindFirstChild("HumanoidRootPart")
    local originPos = myHRP and (myHRP.Position + Vector3.new(0, 1.5, 0))
        or Workspace.CurrentCamera.CFrame.Position

    local targetPart = getClosestTarget(originPos, CurrentAimContext)
    if not targetPart then
        for _, p in ipairs(snakeAuraParts) do
            if p and p.Parent then p:Destroy() end
        end
        snakeAuraParts = {}
        return
    end

    while #snakeAuraParts < snakeAuraSegments do
        local p = Instance.new("Part")
        p.Name = "SnakeAura"
        p.Anchored = true
        p.CanCollide = false
        p.CanQuery = false
        p.CanTouch = false
        p.Massless = true
        p.Material = Enum.Material.Neon
        p.Shape = Enum.PartType.Ball
        p.Size = Vector3.new(snakeAuraSize, snakeAuraSize, snakeAuraSize)
        p.Color = snakeAuraColor
        p.Transparency = 0.2
        p.Parent = Workspace
        table.insert(snakeAuraParts, p)
    end
    while #snakeAuraParts > snakeAuraSegments do
        local p = table.remove(snakeAuraParts)
        if p and p.Parent then p:Destroy() end
    end
    snakeAuraAngle += dt * snakeAuraSpeed
    local basePos = targetPart.Position
    local totalAngle = math.pi * 4
    for i = 1, snakeAuraSegments do
        local p = snakeAuraParts[i]
        local progress = i / snakeAuraSegments
        local angle = snakeAuraAngle + progress * totalAngle
        local heightOffset = (progress - 0.5) * snakeAuraHeight
        local x = math.cos(angle) * snakeAuraRadius
        local z = math.sin(angle) * snakeAuraRadius
        p.Position = basePos + Vector3.new(x, heightOffset, z)
        p.Color = snakeAuraColor
        p.Size = Vector3.new(snakeAuraSize, snakeAuraSize, snakeAuraSize)
    end
end)

RunService.Heartbeat:Connect(function(dt)
    if not CFG.Enabled then return end

    local myChar = LocalPlayer.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not myRoot then return end

    local now = tick()

    if now - wouldLastSpawn >= CFG.SpawnRate and #wouldTrails < CFG.MaxTrails then
        spawnTrail(myRoot.Position)
        wouldLastSpawn = now
    end

    for i = #wouldTrails, 1, -1 do
        local t = wouldTrails[i]

        if now - t.born >= CFG.LifeTime then
            removeWouldTrail(t)
            table.remove(wouldTrails, i)
            continue
        end

        local move  = CFG.Speed * dt
        t.traveled += move
        t.headPos   = t.headPos + t.dir * move

        t.segments[#t.segments].head.Position = t.headPos

        if t.traveled >= CFG.TurnInterval then
            t.traveled = 0
            t.dir = pickDir(t.dir)

            local newTail = makePart(t.headPos)
            local newHead = makePart(t.headPos)
            local newBeam = makeBeam(newTail, newHead)
            table.insert(t.segments, { head = newHead, tail = newTail, beam = newBeam })

            if #t.segments > CFG.MaxSegs then
                local old = table.remove(t.segments, 1)
                pcall(function() old.head:Destroy(); old.tail:Destroy() end)
            end

            for j, seg in ipairs(t.segments) do
                seg.beam.Transparency = NumberSequence.new(1 - (j / #t.segments))
            end
        end
    end
end)

task.spawn(function()
    while true do
        task.wait(COOLDOWN.TriggerBotDelay)

        local myHRP = LocalPlayer.Character
            and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
        local origin = myHRP and (myHRP.Position + Vector3.new(0, 1.5, 0)) or nil
        if not origin then continue end

        if Settings.GunTriggerBot then
            local gun = getGun()
            if gun and canShoot(gun) and getClosestTarget(origin, "Gun") then
                local fsm = getFSM(gun)
                if fsm then
                    CurrentAimContext = "Gun"
                    pcall(function() fsm:Set("Fire") end)
                end
            end
        end

        if Settings.KnifeTriggerBot and tick() >= Cooldowns.Throw then
            local knife = getKnife()
            local fsm = knife and getFSM(knife)
            if fsm and getClosestTarget(origin, "Knife") then
                CurrentAimContext = "Knife"
                local ok = pcall(function() return fsm:Set("Throw") end)
                if ok then
                    Cooldowns.Throw = tick() + COOLDOWN.Throw
                end
            end
        end
    end
end)
